### CRM OAuth Registry (Hub)

**File**: src/services/crm/CrmOAuthRegistry.ts

\\\	ypescript
import { SalesforceOAuthProvider } from './salesforce/SalesforceOAuthProvider';
import { HubSpotOAuthProvider } from './hubspot/HubSpotOAuthProvider';
import { PipedriveOAuthProvider } from './pipedrive/PipedriveOAuthProvider';
import { ICrmOAuthProvider } from './interfaces/ICrmOAuthProvider';

type CrmProvider = 'salesforce' | 'hubspot' | 'pipedrive';

export class CrmOAuthRegistry {
  private providers: Map<CrmProvider, ICrmOAuthProvider>;

  constructor() {
    this.providers = new Map();
    this.registerProviders();
  }

  private registerProviders(): void {
    this.providers.set('salesforce', new SalesforceOAuthProvider());
    this.providers.set('hubspot', new HubSpotOAuthProvider());
    this.providers.set('pipedrive', new PipedriveOAuthProvider());
  }

  getProvider(name: CrmProvider): ICrmOAuthProvider {
    const provider = this.providers.get(name);
    if (!provider) {
      throw new Error(\CRM provider \ not found\);
    }
    return provider;
  }

  getAvailableProviders(): CrmProvider[] {
    return Array.from(this.providers.keys());
  }
}

export const crmOAuthRegistry = new CrmOAuthRegistry();

### CRM Routes (Minimal Week 1-2 Implementation)

**File**: src/routes/crm/index.ts

\\\	ypescript
import { Router, Request, Response } from 'express';
import { crmOAuthRegistry } from '@services/crm/CrmOAuthRegistry';
import { CrmUserProvisioner } from '@services/crm/CrmUserProvisioner';
import { v4 as uuidv4 } from 'uuid';

const crmRouter = Router();
const provisioner = new CrmUserProvisioner();

// CRM OAuth initiation
crmRouter.get('/authorize/:provider', async (req: Request, res: Response) => {
  try {
    const { provider } = req.params as { provider: 'salesforce' | 'hubspot' | 'pipedrive' };
    const state = uuidv4();
    
    // Store state in session/cache (TODO: implement state manager)
    const oauthProvider = crmOAuthRegistry.getProvider(provider);
    const authUrl = await oauthProvider.getAuthorizationUrl(state);
    
    res.redirect(authUrl);
  } catch (error) {
    res.status(400).json({ error: 'Authorization failed' });
  }
});

// CRM OAuth callback
crmRouter.get('/callback/:provider', async (req: Request, res: Response) => {
  try {
    const { provider } = req.params as { provider: 'salesforce' | 'hubspot' | 'pipedrive' };
    const { code, state } = req.query as { code: string; state: string };
    
    // TODO: Validate state
    const oauthProvider = crmOAuthRegistry.getProvider(provider);
    const token = await oauthProvider.exchangeCodeForToken(code);
    const crmUser = await oauthProvider.getUserProfile(token.accessToken);
    
    // Provision user in ProspectPI
    const prospectpiUser = await provisioner.provisionFromCRM(provider, crmUser, token);
    
    res.json({
      success: true,
      userId: prospectpiUser.id,
      crmProvider: provider
    });
  } catch (error) {
    res.status(400).json({ error: 'Callback handling failed' });
  }
});

export default crmRouter;
