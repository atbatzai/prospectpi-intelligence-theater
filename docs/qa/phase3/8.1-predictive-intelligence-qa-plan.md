import { PredictiveIntelligenceEngine } from '../PredictiveIntelligenceEngine';
import { mockDossier } from '../../../tests/fixtures/dossiers';

describe('PredictiveIntelligenceEngine', () => {
  let engine: PredictiveIntelligenceEngine;

  beforeEach(async () => {
    engine = new PredictiveIntelligenceEngine();
    await engine.initialize();
  });

  describe('AC2: Real-time Prediction', () => {
    test('should generate predictions within 2 seconds', async () => {
      const start = Date.now();
      const prediction = await engine.predict(mockDossier);
      const latency = Date.now() - start;

      expect(latency).toBeLessThan(2000);
      expect(prediction).toBeDefined();
    });

    test('should return all required prediction fields', async () => {
      const prediction = await engine.predict(mockDossier);

      expect(prediction.fundingLikelihood).toBeGreaterThanOrEqual(0);
      expect(prediction.fundingLikelihood).toBeLessThanOrEqual(100);
      expect(prediction.confidence).toBeGreaterThanOrEqual(0);
      expect(prediction.confidence).toBeLessThanOrEqual(100);
      expect(prediction.modelVersion).toBeDefined();
      expect(prediction.timestamp).toBeInstanceOf(Date);
    });
  });

  describe('AC3: Prediction Categories', () => {
    test('should provide all 5 prediction categories', async () => {
      const prediction = await engine.predict(mockDossier);

      expect(prediction).toHaveProperty('fundingLikelihood');
      expect(prediction).toHaveProperty('hiringSignals');
      expect(prediction).toHaveProperty('techAdoption');
      expect(prediction).toHaveProperty('marketExpansion');
      expect(prediction).toHaveProperty('competitiveVulnerability');
    });

    test('should include top 3 signals', async () => {
      const prediction = await engine.predict(mockDossier);

      expect(prediction.topSignals).toHaveLength(3);
      expect(Array.isArray(prediction.topSignals)).toBe(true);
    });
  });

  describe('AC5: Explainable AI', () => {
    test('should provide top 5 contributing factors', async () => {
      const prediction = await engine.predict(mockDossier);

      expect(prediction.explanation.topFactors).toHaveLength(5);
      prediction.explanation.topFactors.forEach(factor => {
        expect(factor).toHaveProperty('factor');
        expect(factor).toHaveProperty('impact');
        expect(factor).toHaveProperty('citation');
      });
    });

    test('should include confidence intervals', async () => {
      const prediction = await engine.predict(mockDossier);

      expect(prediction.explanation.confidenceInterval).toBeDefined();
      expect(prediction.explanation.confidenceInterval.lower).toBeLessThan(
        prediction.explanation.confidenceInterval.upper
      );
    });
  });

  describe('AC4: Performance Monitoring', () => {
    test('should log predictions to database', async () => {
      const prediction = await engine.predict(mockDossier);
      
      // Verify logging occurred (check database)
      // TODO: Add database verification
      expect(prediction.dossierId).toBe(mockDossier.id);
    });

    test('should track model metrics', async () => {
      const metrics = await engine.getModelMetrics();

      expect(metrics.accuracy).toBeGreaterThanOrEqual(0.8);
      expect(metrics.version).toBeDefined();
      expect(metrics.isActive).toBe(true);
    });
  });
});

// tests/performance/predictions-load-test.js
import http from 'k6/http';
import { check, sleep } from 'k6';

export let options = {
  stages: [
    { duration: '1m', target: 50 },   // Ramp up to 50 users
    { duration: '3m', target: 50 },   // Stay at 50 users
    { duration: '1m', target: 100 },  // Ramp to 100 users
    { duration: '3m', target: 100 },  // Stay at 100 users
    { duration: '1m', target: 0 },    // Ramp down
  ],
  thresholds: {
    http_req_duration: ['p(95)<2000'], // 95% of requests < 2s
    http_req_failed: ['rate<0.01'],     // <1% errors
  },
};

export default function () {
  const dossierId = 'test-dossier-id';
  const response = http.post(\http://localhost:3001/api/v1/predictions/\\, null, {
    headers: { 'Authorization': 'Bearer test_token' },
  });

  check(response, {
    'status is 200': (r) => r.status === 200,
    'response time < 2s': (r) => r.timings.duration < 2000,
    'has prediction data': (r) => JSON.parse(r.body).prediction !== undefined,
  });

  sleep(1);
}
